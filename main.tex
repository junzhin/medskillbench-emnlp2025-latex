% Chart placeholder: faux axes + bar silhouette.
\newcommand{\chartbox}[2]{%
  \begin{center}
  \begin{tikzpicture}
    \def\W{\linewidth}
    \node[draw=black!60, thick, minimum width=\W, minimum height=#1,
          inner sep=0pt, fill=orange!3] (b) {};
    % faux axes inside
    \pgfmathsetlengthmacro{\inW}{0.78\linewidth}
    \pgfmathsetlengthmacro{\inH}{#1-1.0cm}
    \coordinate (origin) at ([xshift=0.10\linewidth, yshift=-0.35cm]b.south west);
    \draw[black!70, thick] (origin) -- ++(\inW,0);
    \draw[black!70, thick] (origin) -- ++(0,\inH);
    % bars
    \foreach \i/\h in {0/0.45,1/0.78,2/0.32,3/0.91,4/0.55,5/0.68}{
      \pgfmathsetlengthmacro{\bx}{\i*0.12\linewidth + 0.03\linewidth}
      \pgfmathsetlengthmacro{\bh}{\h*\inH}
      \fill[orange!60, draw=orange!80] ([xshift=\bx]origin) rectangle ++(0.08\linewidth,\bh);
    }
    \node[fill=white, draw=black!50, inner sep=3pt, font=\scriptsize\itshape,
          align=center, text width=0.9\linewidth, below=2pt of b.north]
          {\textbf{[CHART PLACEHOLDER]} \ #2};
  \end{tikzpicture}
  \end{center}
}
